% A4 paper size by default, use 'letterpaper' for US letter

% Configure page margins with geometry
% footskip must stay above ~13.6pt or fancyhdr drives the footer into the last line.
\geometry{left=1.0cm, top=1.0cm, right=1.0cm, bottom=0.8cm, footskip=.55cm}

% Color for highlights - Black and white theme
\definecolor{awesome}{HTML}{000000}

% Set false if you don't want to highlight section with awesome color
\setbool{acvSectionColorHighlight}{false}

% \parskip's default differs between TeX Live versions, and the class loads parskip without
% pinning it. Left alone, every paragraph break on the page is worth a different amount here
% than on the GitHub runner -- about 80pt over a full CV, which is why the published PDF and
% the local one disagreed on how much of the page was used. Pin it so both builds agree.
\setlength{\parskip}{0pt}

% Section and entry spacing. These are the levers for filling the page: the content
% is fixed at one page, so the slack goes here rather than into a bottom margin.
\renewcommand{\acvSectionTopSkip}{4.5mm}
\renewcommand{\acvSectionContentTopSkip}{1.5mm}
\renewcommand{\acvEntryTopSkip}{3.3mm}

% If you would like to change the social information separator from a pipe (|) to something else
